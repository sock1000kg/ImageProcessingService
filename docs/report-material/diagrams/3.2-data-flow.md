# Hình 3.2: Luồng dữ liệu end-to-end của một batch

plantuml
@startuml
title Hình 3.2: Luồng dữ liệu end-to-end của một batch
skinparam defaultFontName Arial
skinparam ActivityBorderColor #555555
skinparam SwimlaneTitleFontStyle bold

|Trình duyệt|
start
:Tải file lên
POST /api/batches/:id/files
(video: multipart 8 MB/phần);

|Backend|
:Lưu file gốc vào MinIO
uploads/<batchId>/<jobId>/input.<ext>;
:Ghi PostgreSQL: Job **PENDING**;

|Trình duyệt|
:Bấm chạy
POST /api/batches/:id/start;

|Backend|
:publishJob → jobs.image / jobs.video
Job **QUEUED**;

fork
  |NATS JetStream|
  :Lưu message vào stream
  IMAGE_JOBS / VIDEO_JOBS;

  |KEDA|
  :Thấy lag của consumer
  scale worker 0 → N;

  |Worker|
  :Tải input, phát hiện,
  whitelist, che, xoá EXIF;

  fork
    :Ghi kết quả lên MinIO
    results/<batchId>/<jobId>/result.<ext>;
  fork again
    :Phát jobs.events
    PROCESSING → COMPLETED / FAILED;
  end fork

  |Backend|
  :applyJobEvent
  cập nhật Job (idempotent);
fork again
  |Trình duyệt|
  :Polling
  GET /api/batches/:id;
end fork

|Trình duyệt|
:Xem bằng presigned URL
(kết quả 1 giờ, ảnh gốc 5 phút);
stop
@enduml
